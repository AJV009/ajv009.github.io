{{- /* Sub-Section Shortcode Creates a collapsible section with glass morphism styling matching series/TOC sections. Supports two modes: 1. FILE MODE - Load content from a markdown file: {{< sub-section "assets/file.md" >}} {{< sub-section "assets/file.md" icon="fa-code" >}} 2. INLINE MODE - Use inline content: {{< sub-section title="My Title" >}} Your markdown content here... {{< /sub-section >}} {{< sub-section title="Code Example" icon="fa-code" >}} ```javascript console.log("hello"); ``` {{< /sub-section >}} Parameters: - Positional 0: File path (for file mode) - title: Title for inline mode (or override for file mode) - icon: FontAwesome icon class (default: "fa-book-open") */ -}} {{- $filePath := .Get 0 -}} {{- $currentPage := .Page -}} {{- $title := "" -}} {{- $content := "" -}} {{- $isFileMode := false -}} {{- $icon := .Get "icon" | default "fa-book-open" -}} {{- /* Determine mode: file-based or inline */ -}} {{- if $filePath -}} {{- /* FILE MODE: Load content from file */ -}} {{- $resource := $currentPage.Resources.GetMatch $filePath -}} {{- if $resource -}} {{- $isFileMode = true -}} {{- $content = $resource.Content | safeHTML -}} {{- $title = $resource.Params.title | default "Additional Resource" -}} {{- end -}} {{- else -}} {{- /* INLINE MODE: Use provided title and inner content */ -}} {{- $isFileMode = true -}} {{- $title = .Get "title" | default "Additional Content" -}} {{- $content = .Inner | .Page.RenderString -}} {{- end -}} {{- if $isFileMode -}} {{- /* Generate a unique ID for this section */ -}} {{- $sectionId := anchorize $title -}} {{- $iconClass := printf "sub-section-icon-%s" $sectionId -}}
{{ $content }}
{{- else -}} {{- if $filePath -}} {{- /* File was specified but not found */ -}}

Error: Could not find resource at {{ $filePath }}

{{- else -}} {{- /* No file and no inline content */ -}}

Warning: Sub-section shortcode requires either a file path or inline content with a title parameter.

{{- end -}} {{- end -}}
